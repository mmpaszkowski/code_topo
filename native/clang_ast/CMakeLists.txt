cmake_minimum_required(VERSION 3.20)
project(indexer_clang_ast LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
if(NOT LLVM_VERSION_MAJOR EQUAL 21)
  message(FATAL_ERROR "The extractor requires LLVM/Clang 21")
endif()
find_package(Clang REQUIRED CONFIG HINTS "${LLVM_INSTALL_PREFIX}/lib/cmake/clang")
add_executable(indexer-clang-ast main.cpp)
target_compile_features(indexer-clang-ast PRIVATE cxx_std_17)
target_include_directories(indexer-clang-ast SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
target_link_libraries(indexer-clang-ast PRIVATE clang-cpp LLVM)
